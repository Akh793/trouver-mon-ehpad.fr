# Accès ouvert, y compris aux robots des moteurs de réponse — GPTBot,
# OAI-SearchBot, ChatGPT-User, ClaudeBot, PerplexityBot, Google-Extended,
# Applebot-Extended, Bingbot. Aucune règle ne les vise nommément : sans règle,
# un robot est autorisé. Les déclarer dans des groupes séparés serait pire
# qu'inutile — un groupe nommé prend le pas sur « User-agent: * », et ces
# robots cesseraient alors d'hériter des deux interdictions ci-dessous.

User-agent: *
Allow: /

# Deux familles d'adresses à paramètres, qu'aucun moteur ne doit explorer :
#  — les combinaisons de filtres du calculateur, qui ne créent aucune page
#    distincte et n'ont donc rien à indexer ;
#  — les liens de partage, qui transportent la situation de la personne qui
#    les a produits — ressources, GIR, composition du foyer. Ils sont destinés
#    à un proche, pas à un résultat de recherche.
Disallow: /*?
Allow: /$
